

mkdir -p ~/.ssh
chmod 700 ~/.ssh

# Generate the guest's own keypair. build.py exports ~/.ssh/id_rsa.pub as
# <output>-id_rsa.pub (the key the VM uses to connect back to the build host
# for rsync), so it must exist before build.py reads it.
rm -f ~/.ssh/id_rsa
ssh-keygen -t rsa -f ~/.ssh/id_rsa -q -N ''

# Restore the SELinux contexts on everything we just created. This family
# runs SELinux enforcing, and a ~/.ssh written by a plain shell here inherits
# the wrong label, which makes sshd refuse the key on the NEXT boot -- the
# same failure hooks/host_prepareImage.sh relabels for at image-prep time,
# reintroduced after the fact. restorecon is a no-op when SELinux is
# disabled, so this is safe unconditionally.
command -v restorecon >/dev/null 2>&1 && restorecon -R ~/.ssh

# Do NOT reconfigure or restart sshd here. Root key login is already set up
# by hooks/host_prepareImage.sh (PermitRootLogin / PubkeyAuthentication /
# AcceptEnv are appended to /etc/ssh/sshd_config before first boot), and
# restarting sshd from inside the session build.py is currently driving can
# drop the connection mid-build.
